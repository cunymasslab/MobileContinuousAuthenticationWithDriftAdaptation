def sha256(path: Path) -> str:
    digest = hashlib.sha256()

    with path.open("rb") as handle:
        for block in iter(
            lambda: handle.read(1024 * 1024),
            b"",
        ):
            digest.update(block)

    return digest.hexdigest()


def bootstrap_mean_ci(
    values: np.ndarray,
    seed: int,
    n_boot: int = N_BOOT,
) -> tuple[float, float]:
    values = np.asarray(values, dtype=float)
    values = values[np.isfinite(values)]

    if len(values) < 2:
        return np.nan, np.nan

    rng = np.random.default_rng(seed)
    boot_means = np.empty(n_boot, dtype=float)
    n = len(values)

    for start in range(0, n_boot, BOOT_BATCH):
        stop = min(start + BOOT_BATCH, n_boot)
        batch_size = stop - start

        indices = rng.integers(
            0,
            n,
            size=(batch_size, n),
        )

        boot_means[start:stop] = (
            values[indices].mean(axis=1)
        )

    return (
        float(np.quantile(boot_means, 0.025)),
        float(np.quantile(boot_means, 0.975)),
    )


def cluster_weighted_bootstrap_ci(
    cluster_sums: np.ndarray,
    cluster_counts: np.ndarray,
    seed: int,
    n_boot: int = N_BOOT,
) -> tuple[float, float]:
    cluster_sums = np.asarray(cluster_sums, dtype=float)
    cluster_counts = np.asarray(cluster_counts, dtype=float)

    if (
        len(cluster_sums) < 2
        or len(cluster_sums) != len(cluster_counts)
    ):
        return np.nan, np.nan

    rng = np.random.default_rng(seed)
    n_clusters = len(cluster_sums)
    boot_means = np.empty(n_boot, dtype=float)

    for start in range(0, n_boot, BOOT_BATCH):
        stop = min(start + BOOT_BATCH, n_boot)
        batch_size = stop - start

        indices = rng.integers(
            0,
            n_clusters,
            size=(batch_size, n_clusters),
        )

        sampled_sums = cluster_sums[indices].sum(axis=1)
        sampled_counts = cluster_counts[indices].sum(axis=1)

        boot_means[start:stop] = (
            sampled_sums / sampled_counts
        )

    return (
        float(np.quantile(boot_means, 0.025)),
        float(np.quantile(boot_means, 0.975)),
    )


def sign_flip_pvalue(
    differences: np.ndarray,
    seed: int,
    n_resamples: int = N_SIGN_FLIPS,
) -> float:
    differences = np.asarray(differences, dtype=float)
    differences = differences[np.isfinite(differences)]

    if len(differences) == 0:
        return np.nan

    observed = abs(float(differences.mean()))
    rng = np.random.default_rng(seed)

    extreme = 0
    completed = 0
    n = len(differences)

    while completed < n_resamples:
        batch_size = min(
            SIGN_BATCH,
            n_resamples - completed,
        )

        signs = rng.choice(
            np.array([-1.0, 1.0]),
            size=(batch_size, n),
        )

        permuted = np.abs(
            (signs * differences).mean(axis=1)
        )

        extreme += int(
            np.sum(permuted >= observed - 1e-15)
        )

        completed += batch_size

    # Add-one correction for a Monte Carlo randomization test.
    return float(
        (extreme + 1)
        / (n_resamples + 1)
    )


def holm_adjust(pvalues: list[float]) -> list[float]:
    values = np.asarray(pvalues, dtype=float)
    adjusted = np.full(len(values), np.nan)

    valid = np.flatnonzero(np.isfinite(values))

    if len(valid) == 0:
        return adjusted.tolist()

    order = valid[np.argsort(values[valid])]
    running = 0.0
    m = len(order)

    for rank, index in enumerate(order):
        candidate = values[index] * (m - rank)
        running = max(running, candidate)
        adjusted[index] = min(running, 1.0)

    return adjusted.tolist()


def summarize_user_view(
    frame: pd.DataFrame,
    view_name: str,
    seed: int,
) -> dict:
    required = [
        "M6_overall_eer",
        "M7_overall_eer",
        "M8_overall_eer",
    ]

    missing = [
        column
        for column in required
        if column not in frame.columns
    ]

    if missing:
        fail(
            f"{view_name} is missing columns: {missing}"
        )

    m6 = frame["M6_overall_eer"].to_numpy(
        dtype=float
    )
    m7 = frame["M7_overall_eer"].to_numpy(
        dtype=float
    )
    m8 = frame["M8_overall_eer"].to_numpy(
        dtype=float
    )

    differences = m7 - m8
    rtrl_benefit = m6 - m7
    bptt_benefit = m6 - m8

    n = len(differences)

    if n < 2:
        fail(
            f"{view_name} has fewer than two users"
        )

    mean_difference = float(differences.mean())
    median_difference = float(
        np.median(differences)
    )
    sd_difference = float(
        differences.std(ddof=1)
    )
    se_difference = (
        sd_difference / np.sqrt(n)
    )

    critical = student_t.ppf(
        0.975,
        df=n - 1,
    )

    t_ci_low = float(
        mean_difference - critical * se_difference
    )
    t_ci_high = float(
        mean_difference + critical * se_difference
    )

    boot_low, boot_high = bootstrap_mean_ci(
        differences,
        seed=seed,
    )

    t_result = ttest_1samp(
        differences,
        popmean=0.0,
        nan_policy="raise",
    )

    try:
        wilcoxon_result = wilcoxon(
            differences,
            zero_method="wilcox",
            correction=False,
            alternative="two-sided",
            mode="auto",
        )

        wilcoxon_statistic = float(
            wilcoxon_result.statistic
        )
        wilcoxon_p = float(
            wilcoxon_result.pvalue
        )

    except ValueError:
        wilcoxon_statistic = np.nan
        wilcoxon_p = np.nan

    randomization_p = sign_flip_pvalue(
        differences,
        seed=seed + 1000,
    )

    if sd_difference > 0:
        cohens_dz = (
            mean_difference / sd_difference
        )
    else:
        cohens_dz = np.nan

    m7_lower = int(
        np.sum(differences < -TOL)
    )
    m8_lower = int(
        np.sum(differences > TOL)
    )
    ties = int(
        np.sum(np.abs(differences) <= TOL)
    )

    rtrl_ci_low, rtrl_ci_high = bootstrap_mean_ci(
        rtrl_benefit,
        seed=seed + 2000,
    )

    bptt_ci_low, bptt_ci_high = bootstrap_mean_ci(
        bptt_benefit,
        seed=seed + 3000,
    )

    return {
        "view": view_name,
        "n_users": n,
        "M6_mean_eer": float(m6.mean()),
        "M7_mean_eer": float(m7.mean()),
        "M8_mean_eer": float(m8.mean()),
        "mean_M7_minus_M8": mean_difference,
        "mean_difference_percentage_points": (
            100.0 * mean_difference
        ),
        "median_M7_minus_M8": median_difference,
        "sd_M7_minus_M8": sd_difference,
        "se_M7_minus_M8": se_difference,
        "t_ci95_low": t_ci_low,
        "t_ci95_high": t_ci_high,
        "bootstrap_ci95_low": boot_low,
        "bootstrap_ci95_high": boot_high,
        "paired_t_statistic": float(
            t_result.statistic
        ),
        "paired_t_pvalue": float(
            t_result.pvalue
        ),
        "wilcoxon_statistic": wilcoxon_statistic,
        "wilcoxon_pvalue": wilcoxon_p,
        "sign_flip_pvalue": randomization_p,
        "cohens_dz": float(cohens_dz),
        "users_M7_lower": m7_lower,
        "users_M8_lower": m8_lower,
        "users_tied": ties,
        "mean_RTRL_benefit_M6_minus_M7": float(
            rtrl_benefit.mean()
        ),
        "RTRL_benefit_boot_ci95_low": rtrl_ci_low,
        "RTRL_benefit_boot_ci95_high": rtrl_ci_high,
        "mean_BPTT_benefit_M6_minus_M8": float(
            bptt_benefit.mean()
        ),
        "BPTT_benefit_boot_ci95_low": bptt_ci_low,
        "BPTT_benefit_boot_ci95_high": bptt_ci_high,
    }
